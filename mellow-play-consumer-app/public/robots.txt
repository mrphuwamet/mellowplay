# The public side of the site (home, explore, classes, ข่าวสาร/เรื่องน่ารู้) is
# open to crawlers. Everything below Disallow is either a personal account
# page with nothing to rank, or a URL whose path contains a secret: a check-in
# QR, an unlisted album share link, an invite, a certificate code. Those are
# handed to one person and must not end up in a search result.

User-agent: *
Allow: /

# Links that carry a token — the address itself is the permission.

Disallow: /invite/
Disallow: /round/
Disallow: /shared-albums/
Disallow: /unsubscribe/
Disallow: /verify/
Disallow: /certificate/
Disallow: /report/
Disallow: /reset-password

# Signed-in areas. Nothing here renders without an account, so a crawler only
# ever reaches a login screen and reports a pile of near-identical pages.

Disallow: /my-coupons
Disallow: /journey
Disallow: /add-child
Disallow: /booking
